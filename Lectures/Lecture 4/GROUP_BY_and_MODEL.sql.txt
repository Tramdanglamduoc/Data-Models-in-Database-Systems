CREATE TABLE bus_drivers (
    driver_first_name varchar2(50),
    driver_last_name varchar2(50) NOT NULL,
    personal_code varchar2(11) PRIMARY KEY
);

CREATE TABLE salaries (
    paid_salary number(6,2),
    hours_driven number(3),
    year number(4),
    month char(3),
    driver_code varchar2(11),
    CONSTRAINT pk_salaries PRIMARY KEY (year, month, driver_code),
    CONSTRAINT fk_driver_salaries FOREIGN KEY (driver_code) 
        REFERENCES bus_drivers(personal_code) ON DELETE CASCADE
);

CREATE TABLE buses (
    bus_number varchar2(8) PRIMARY KEY,
    seat_count number(3),
    availability char(1),
    cost_per_kilometer number(3,2),
    hourly_rental_cost number(4,2)
);

CREATE TABLE trips (
    trip_number number(3) PRIMARY KEY,
    departure_time date,
    duration number(4),
    destination varchar2(30),
    kilometers number,
    driver varchar2(11),
    bus_number varchar2(8),
    CONSTRAINT fk_trip_driver FOREIGN KEY (driver) 
        REFERENCES bus_drivers(personal_code) ON DELETE SET NULL,
    CONSTRAINT fk_trip_bus FOREIGN KEY (bus_number) 
        REFERENCES buses(bus_number) ON DELETE SET NULL
);

--PIVOT command
--Querying trips directly gives a chaotic result
SELECT * FROM trips
PIVOT(
    COUNT(trip_number)
    FOR destination
    IN ('Berlin', 'Madrid')
) 
ORDER BY bus_number;

--This is one of the purposes for using views
--Create a view with only the fields we need
CREATE OR REPLACE VIEW trip_statistics AS
(SELECT trip_number, destination, bus_number FROM trips);

SELECT * FROM trip_statistics
PIVOT(
    COUNT(trip_number)
    FOR destination
    IN ('Berlin', 'Madrid', 'Poznan', 'Rome', 'Prague')
) 
ORDER BY bus_number;

--GROUP BY options
SELECT destination, bus_number, driver, count(trip_number) AS COUNT
FROM trips
GROUP BY destination, bus_number, driver;

SELECT destination, bus_number, driver, count(trip_number) 
FROM trips
GROUP BY ROLLUP (destination, bus_number, driver);

SELECT destination, bus_number, driver, count(trip_number) 
FROM trips
GROUP BY CUBE (destination, bus_number, driver);

--GROUPING creates a bitmap

SELECT destination, bus_number, driver, count(trip_number), 
    GROUPING(destination) as d,  
    GROUPING(bus_number) as bn, 
    GROUPING(driver) as dr
FROM trips
GROUP BY CUBE (destination, bus_number, driver);

--We can also use HAVING with these groups

SELECT destination, bus_number, driver, count(trip_number), 
    GROUPING(destination) as d,  
    GROUPING(bus_number) as bn, 
    GROUPING(driver) as dr
FROM trips
GROUP BY CUBE (destination, bus_number, driver)
HAVING GROUPING(destination) = 0;

--To format results, we use DECODE
SELECT destination, bus_number, driver, count(trip_number), 
    DECODE(GROUPING(destination),1,'yes',0,'no') as d,  
    DECODE(GROUPING(bus_number),1,'yes', 0, 'no') as bn, 
    DECODE(GROUPING(driver),1,'yes') as dr
FROM trips
GROUP BY CUBE (destination, bus_number, driver)
HAVING GROUPING(destination) = 0;

SELECT 
    DECODE(GROUPING(destination),1,'Total', destination) as d,  
    DECODE(GROUPING(bus_number),1,'Total', bus_number) as bn, 
    DECODE(GROUPING(driver),1,'Total', driver) as dr, 
    count(trip_number)
FROM trips
GROUP BY CUBE (destination, bus_number, driver)
HAVING GROUPING(destination) = 0;

--Why doesn't it work directly?
--MODEL doesn't group, it works with existing values
--Let’s create a grouped view

CREATE OR REPLACE VIEW trip_statistics_model AS
(SELECT driver, destination, bus_number, count(trip_number) as trip_count
 FROM trips
 GROUP BY driver, destination, bus_number);

SELECT * 
FROM trip_statistics_model
MODEL 
DIMENSION BY (destination, driver, bus_number)  
MEASURES (trip_count)
RULES(); 	

--To demonstrate PARTITION BY, let’s use the salaries table,
--where the primary key is "naturally" structured
--here we have an error which comes from logics - months would repeat
SELECT * 
FROM salaries s JOIN bus_drivers d 
ON s.driver_code = d.personal_code
MODEL 
DIMENSION BY (year, driver_last_name)  
MEASURES (paid_salary)
RULES(); 	

SELECT * 
FROM salaries s JOIN bus_drivers d 
ON s.driver_code = d.personal_code
MODEL 
PARTITION BY (month)
DIMENSION BY (year, driver_last_name)  
MEASURES (paid_salary)
RULES()
ORDER BY year;

--RULES section

--Without PARTITION
SELECT * 
FROM salaries s JOIN bus_drivers d 
ON s.driver_code = d.personal_code
MODEL 
DIMENSION BY (year, driver_last_name, month)  
MEASURES (paid_salary)
RULES(
    paid_salary[2023,'Adkins','JAN'] = paid_salary[2023,'Adkins','JAN']*1.1
)
ORDER BY year;

--What happens with PARTITION?
SELECT * 
FROM salaries s JOIN bus_drivers d 
ON s.driver_code = d.personal_code
MODEL 
PARTITION BY (month)
DIMENSION BY (year, driver_last_name)  
MEASURES (paid_salary)
RULES(
    paid_salary[2023,'Adkins'] = paid_salary[2023,'Adkins']*1.1
)
ORDER BY year;

--Various rule examples
SELECT * 
FROM salaries s JOIN bus_drivers d 
ON s.driver_code = d.personal_code
MODEL 
DIMENSION BY (year, driver_last_name, month)  
MEASURES (paid_salary)
RULES(
    paid_salary[2023,'Adkins','JAN'] = paid_salary[2023,'Adkins','JAN']*1.1,
    paid_salary[year > 2021,'Knowles','DEC'] = paid_salary[2023,'Knowles','JAN']*1.1,
    paid_salary[2023,'Bowie','JAN'] = MAX(paid_salary)[2023,'Adkins', month BETWEEN 'NOV' AND 'DEC'],
    paid_salary[2022, 'Mercury', month IN ('NOV', 'DEC')] = paid_salary[2022, 'Freidenfelds', CV(month)]*1.5,
    paid_salary[2022, driver_last_name IS ANY, 'OCT'] = 10000
)
ORDER BY year;
