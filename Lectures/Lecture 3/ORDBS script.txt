CREATE TYPE ADDRESS_T AS OBJECT(
   STREET    VARCHAR2(20),
   A_NUMBER   NUMBER, 
   CITY  VARCHAR2(20));

CREATE TYPE PERSON_T AS OBJECT ( 
   P_ID       NUMBER,    
   SURNAME  VARCHAR2(15),
   P_NAME     VARCHAR2(15),
   ADDRESS REF ADDRESS_T );

CREATE TABLE ADDRESSES_TABLE OF ADDRESS_T;

CREATE TABLE PEOPLE1 OF PERSON_T
(ADDRESS with ROWID scope IS ADDRESSES_TABLE);

INSERT  INTO ADDRESSES_TABLE VALUES(ADDRESS_T('Stabu', 70, 'Riga'));
